#!/usr/bin/env python3
"""Deterministic final-merge checks for Vietnamese reader copy.

This is a release lint. It reports PASS/WARNING/BLOCKED and never claims a
numeric naturalness score. Source claims are supplied separately so proof is
not inferred from fluent wording.
"""
from __future__ import annotations

import argparse
import json
import re
from pathlib import Path

SCHEMA = "viet-natural-release-lint/v1"

GENERIC_ENGLISH = {
    "workflow", "brief", "research", "status", "scope", "approval", "draft",
    "output", "input", "handoff", "native", "qa", "summary", "insight",
    "framework", "feedback", "next step", "sign off",
}
TRANSLATION_CUES = (
    (r"không chỉ\s+.+\s+mà còn", "cấu trúc đối xứng kiểu dịch"),
    (r"khai phá toàn bộ tiềm năng", "khẩu hiệu trừu tượng"),
    (r"một cách\s+(?:liền mạch|toàn diện|hiệu quả|tối ưu)", "cụm trạng từ máy móc"),
    (r"hành trình\s+(?:đột phá|chuyển đổi)", "ẩn dụ chung chung"),
    (r"tạo ra\s+(?:giá trị|tác động)", "động từ trừu tượng"),
    (r"đóng vai trò", "cụm vai trò trừu tượng"),
)
AI_SOUNDING_CUES = (
    (r"tối ưu hóa hiệu quả", "lợi ích trừu tượng; cần nêu kết quả đo được"),
    (r"hệ sinh thái", "ẩn dụ chung; cần nêu các thành phần hoặc quan hệ"),
    (r"đồng hành\s+kiến tạo", "khẩu hiệu thương hiệu; cần nêu chủ thể và việc làm"),
    (r"mang tính\s+(?:chiến lược|toàn diện|đột phá)", "cụm bổ nghĩa trừu tượng"),
    (r"(?:tiếp tục\s+)?theo dõi", "quy trình rỗng; cần nêu ai theo dõi gì và khi nào"),
    (r"(?:đẩy mạnh|tăng cường)\s+(?:hoạt động|nỗ lực|triển khai)", "động từ chung; cần nêu hành động cụ thể"),
    (r"câu hỏi là.+?câu trả lời là", "cấu trúc tự hỏi–tự trả lời"),
    (r"từ\s+[^.!?\n]{2,60}\s+đến\s+[^.!?\n]{2,60}", "khung đối xứng; kiểm tra xem có đang kéo dài ý"),
)
EM_DASH = re.compile(r"—")
PROOF = re.compile(
    r"(?:hơn|trên|gần|khoảng)?\s*[\d][\d.,]*\s*%?\s*"
    r"(?:khách hàng|người dùng|học viên|đánh giá|lượt xem|người theo dõi|"
    r"người tin tưởng|người hài lòng)", re.I)
LOWER_SENTENCE = re.compile(r"(?m)^[ \t]*(?:[-*•]\s*)?[a-zà-ỹđ]")


def _line(text: str, start: int) -> int:
    return text.count("\n", 0, start) + 1


def _normal(text: str) -> str:
    return re.sub(r"\s+", " ", text).strip()


def load_claims(path: Path | None) -> list[str]:
    if path is None:
        return []
    data = json.loads(path.read_text(encoding="utf-8-sig"))
    if isinstance(data, dict):
        data = data.get("claims", [])
    if not isinstance(data, list) or not all(isinstance(x, str) and x.strip() for x in data):
        raise ValueError("source claims must be a JSON list of non-empty strings")
    return [_normal(x).lower() for x in data]


def _proof_supported(match: str, claims: list[str]) -> bool:
    needle = _normal(match).lower()
    return any(needle in _normal(claim).lower() or _normal(claim).lower() in needle for claim in claims)


def lint(text: str, allowed_english: set[str] | None = None, claims: list[str] | None = None) -> dict:
    allowed = {x.casefold().strip() for x in (allowed_english or set()) if x.strip()}
    claims = claims or []
    findings = []

    for match in PROOF.finditer(text):
        if not _proof_supported(match.group(0), claims):
            findings.append({"severity": "BLOCKED", "rule": "source-proof", "line": _line(text, match.start()),
                             "sample": _normal(match.group(0)),
                             "message": "Cần đối chiếu mệnh đề định lượng với nguồn claim."})

    for match in LOWER_SENTENCE.finditer(text):
        findings.append({"severity": "WARNING", "rule": "sentence-opening-case", "line": _line(text, match.start()),
                         "sample": _normal(match.group(0)),
                         "message": "Câu tiếng Việt hoàn chỉnh nên mở đầu bằng chữ hoa."})

    low = text.casefold()
    for term in sorted(GENERIC_ENGLISH, key=len, reverse=True):
        if term in allowed:
            continue
        for match in re.finditer(rf"(?<![\w]){re.escape(term)}(?![\w])", low):
            if any(term in allowed_term and re.search(rf"(?<![\w]){re.escape(allowed_term)}(?![\w])", low[max(0, match.start()-len(allowed_term)):match.end()+len(allowed_term)]) for allowed_term in allowed):
                continue
            findings.append({"severity": "WARNING", "rule": "generic-english", "line": _line(text, match.start()),
                             "sample": term, "message": "Dịch hoặc giải thích nếu không phải tên riêng hay thuật ngữ bắt buộc."})

    for pattern, label in TRANSLATION_CUES:
        for match in re.finditer(pattern, low):
            findings.append({"severity": "WARNING", "rule": "translation-shaped", "line": _line(text, match.start()),
                             "sample": _normal(match.group(0)), "message": label})

    for pattern, label in AI_SOUNDING_CUES:
        for match in re.finditer(pattern, low, re.S):
            findings.append({"severity": "WARNING", "rule": "ai-sounding-signal", "line": _line(text, match.start()),
                             "sample": _normal(match.group(0))[:120], "message": label})

    for match in EM_DASH.finditer(text):
        before = text.count("—", 0, match.start())
        if before >= 1:
            findings.append({"severity": "WARNING", "rule": "punctuation-cadence", "line": _line(text, match.start()),
                             "sample": "—", "message": "Hai gạch ngang dài trong một câu; kiểm tra xem có nên tách câu."})

    # Three comma-separated noun phrases are a review trigger, not an automatic failure.
    for match in re.finditer(r"[^.!?\n]{1,180},[^.!?\n]{1,180},[^.!?\n]+", text):
        findings.append({"severity": "WARNING", "rule": "mechanical-list-rhythm", "line": _line(text, match.start()),
                         "sample": _normal(match.group(0))[:120], "message": "Kiểm tra xem danh sách có đang đều và máy móc không."})

    status = "BLOCKED" if any(x["severity"] == "BLOCKED" for x in findings) else "WARNING" if findings else "PASS"
    return {"schema": SCHEMA, "status": status, "findingCount": len(findings), "findings": findings,
            "limitations": ["Tín hiệu định lượng; không phải điểm tự nhiên hóa.",
                            "PASS không thay thế đọc soát tiếng Việt, kiểm tra nguồn hoặc kiểm tra kết xuất."]}


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("input", type=Path)
    parser.add_argument("--source-claims", type=Path)
    parser.add_argument("--allow-term", action="append", default=[])
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    try:
        text = args.input.read_text(encoding="utf-8-sig")
        result = lint(text, set(args.allow_term), load_claims(args.source_claims))
    except (OSError, UnicodeError, ValueError, json.JSONDecodeError) as exc:
        result = {"schema": SCHEMA, "status": "BLOCKED", "findingCount": 1,
                  "findings": [{"severity": "BLOCKED", "rule": "input", "message": str(exc)}],
                  "limitations": ["Không đọc được đầu vào."]}
    encoded = json.dumps(result, ensure_ascii=False, indent=2) + "\n"
    if args.output:
        args.output.write_text(encoded, encoding="utf-8")
    print(encoded, end="")
    return 2 if result["status"] == "BLOCKED" else 1 if result["status"] == "WARNING" else 0


if __name__ == "__main__":
    raise SystemExit(main())
